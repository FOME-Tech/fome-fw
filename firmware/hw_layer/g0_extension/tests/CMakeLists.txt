cmake_minimum_required(VERSION 3.16)
project(g0_tests LANGUAGES CXX)
enable_testing()
add_compile_options(-Wall -Wextra -Werror -UNDEBUG)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(G0_SANITIZE "Enable host sanitizers" OFF)
if(G0_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_executable(g0_tests test.cpp ../link.cpp)
add_test(NAME g0_tests COMMAND g0_tests)
