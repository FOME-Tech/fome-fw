cmake_minimum_required(VERSION 3.16)
project(g0_tests LANGUAGES CXX)
enable_testing()
add_executable(g0_tests test.cpp test_bootloader.cpp ../link.cpp ../bootloader.cpp)
target_compile_features(g0_tests PRIVATE cxx_std_17)
target_compile_options(g0_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
option(G0_SANITIZE "Enable host sanitizers" OFF)
if(G0_SANITIZE)
  target_compile_options(g0_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(g0_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME g0_tests COMMAND g0_tests)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME g0_image_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_image.py)
set_tests_properties(g0_image_tests PROPERTIES ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
