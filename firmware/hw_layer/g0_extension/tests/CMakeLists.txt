cmake_minimum_required(VERSION 3.16)
project(g0_tests LANGUAGES CXX)
enable_testing()
add_compile_options(-Wall -Wextra -Werror -UNDEBUG)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(G0_SANITIZE "Enable host sanitizers" OFF)
if(G0_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_executable(g0_tests test.cpp test_bootloader.cpp ../bootloader.cpp ../link.cpp)
add_test(NAME g0_tests COMMAND g0_tests)
add_executable(g0_output_tests test_output_gpio.cpp ../link.cpp)
target_include_directories(g0_output_tests PRIVATE mocks)
add_test(NAME g0_output_tests COMMAND g0_output_tests)
add_executable(g0_updater_tests test_updater.cpp ../bootloader.cpp ../link.cpp)
target_include_directories(g0_updater_tests PRIVATE updater_mocks)
add_test(NAME g0_updater_tests COMMAND g0_updater_tests)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME g0_image_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_image.py)
set_tests_properties(g0_image_tests PROPERTIES ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
